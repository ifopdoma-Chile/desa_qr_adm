Ejemplo rápido (curl)

# Crear un QR de tipo URL
curl -X POST https://sapogis.ifop.cl/api/v1/qrs \
  -H "Authorization: Bearer TU_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"title":"Web IFOP","qr_type":"url","content_url":"https://www.ifop.cl","slug":"ifop-web"}'

# Ver escaneos de un QR
curl "https://sapogis.ifop.cl/api/v1/qrs/ifop-web/scans" \
  -H "Authorization: Bearer TU_API_KEY"
